cmake_minimum_required(VERSION 3.25)
project(GraphicsOpenGL VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(glfw3 REQUIRED)
find_package(glm REQUIRED)
find_package(OpenGL REQUIRED)
add_executable(graphics-opengl src/main.cpp src/shader.cpp src/shader.hpp)
target_link_libraries(graphics-opengl PRIVATE glfw glm OpenGL::GL)
