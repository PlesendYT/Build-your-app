cmake_minimum_required(VERSION 3.25)
project(GraphicsVulkan VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Vulkan REQUIRED)
find_package(glfw3 REQUIRED)
find_package(glm REQUIRED)
add_executable(graphics-vulkan src/main.cpp)
target_link_libraries(graphics-vulkan PRIVATE Vulkan::Vulkan glfw glm)
target_include_directories(graphics-vulkan PRIVATE ${Vulkan_INCLUDE_DIRS})
